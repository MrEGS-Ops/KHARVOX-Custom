cmake_minimum_required(VERSION 3.16)
project(KharvoxStandaloneMods LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN YES)
add_library(kharvox_mod_behind_head_wheel SHARED behind-head-wheel/BehindHeadWheel.cpp)
add_library(kharvox_mod_quick_holsters SHARED dynamic-quick-holsters/QuickHolsters.cpp)
foreach(mod kharvox_mod_behind_head_wheel kharvox_mod_quick_holsters)
  target_include_directories(${mod} PRIVATE include)
  set_target_properties(${mod} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mods/vr" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/mods/vr")
endforeach()
enable_testing()
add_executable(kharvox_standalone_mods_tests tests/mod_tests.cpp)
target_include_directories(kharvox_standalone_mods_tests PRIVATE include)
if(UNIX AND NOT APPLE)
  target_link_libraries(kharvox_standalone_mods_tests PRIVATE dl)
endif()
add_test(NAME standalone_mod_abi_and_behaviour COMMAND kharvox_standalone_mods_tests
  $<TARGET_FILE:kharvox_mod_behind_head_wheel> $<TARGET_FILE:kharvox_mod_quick_holsters>)
